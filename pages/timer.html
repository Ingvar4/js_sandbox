    const timer = (deadline) => {
      if (isNaN(+deadline)) {
        console.log('Передано некорректное число');
        return;
      }
      let time = deadline;
      console.log(time);
      const interval = setInterval(() => {
        time -= 1;
        console.log(time);
      }, 1000);
      setTimeout(() => {
        clearInterval(interval);
        console.log('Время истекло');
      }, deadline * 1000);
    };

    const deadline = prompt('На сколько секунд установить таймер?');
    timer(deadline);
