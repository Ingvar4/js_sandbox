Встроенные методы строки

назад

Встроенные методы строки не мутируют исходную строку

toLowerCase() - приведение к нижнему регистру

  
    let str = 'My name is Alexei';
    console.log(str.toLowerCase()); //my name is alexei
    console.log(str); //My name is Alexei

toUpperCase() - приведение к верхнему регистру

  
    let str = 'My name is Alexei';
    console.log(str.toUpperCase()); //MY NAME IS ALEXEI
    console.log(str); //My name is Alexei

indexOf() - поиск подстроки, возвращает индекс

  
    let str = 'My name is Alexei';
    console.log(str.indexOf('i')); //8
  

  
    let userName = 'Alexei';
    let str = `My name is ${userName}`;
    console.log(str.indexOf('el')); //15

includes() - поиск подстроки, возвращает true/false

  
    let userName = 'Alexei';
    let str = `My name is ${userName}`;
    console.log(
      `The string "${userName}" ${
        str.includes(userName) ? "is" : "is not"
      } in the sentense`
    );
    //The string "Alexei is in the sentense

Пример поиска по ключевому слову

  
    const products = [
      'Яйцо столовое',
      'Молоко Молочник',
      'Коровка Молоко',
      'Виноград Изабелла',
    ];

    let search = 'молоко';

    products.forEach((product) => {
      if (product.toLowerCase().includes(search.toLowerCase())) {
        console.log(product);
      }
    });

    //Молоко Молочник
    //Коровка Молоко

slice()

извлекает часть строки и возвращает новую строку без мутации оригинальной

  
    let userName = 'Alexei';
    let str = `My name is ${userName}`;
    console.log(str.slice(1)); //y name is Alexei
    //Два параметра
    //0 - первый включенный порядковый номер символа
    //5- первый уже не включенный порядковый номер символа
    console.log(str.slice(0, 5)); //My na
    console.log(str.slice(2, 9)); // name i (первый включенный - пробел под индексом 2)

Пример с извлечением даты

  
    let dateStr = '2023-10-05';
    let year = dateStr.slice(0, 4); 
    console.log(year); //2023
    let month = dateStr.slice(5, 7);
    console.log(month); //10
    let day = dateStr.slice(8, 10);
    console.log(day); //05
    console.log(year, month, day); //2023 10 05

Пример принятия отрицательного значения slice

  
    let str = 'Hello World!';
    let newStr = str.slice(0, -1); //Удаляем последний символ
    console.log(newStr); //Hello World

split()

разбивает на массив, путём разделения строки указанной подстрокой

  
    let userName = 'Alexei';
    let str = `My name is ${userName}`;
    console.log(str.split(' '));

    // ['My', 'name', 'is', 'Alexei']

join()

Объединяет все элементы массива или массивоподобного объекта в строку

  
    let userName = 'Alexei';
    let str = `My name is ${userName}`;
    console.log(str.split(' ').join(' '));
    //My name is Alexei

trim()

убирает все пробелы в начале и конце строки

  
    let userName = 'Alexei';
    let str = `   My name is ${userName}  `;
    console.log(str.trim());
    //My name is Alexei
