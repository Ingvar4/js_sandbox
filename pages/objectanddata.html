назад 
  
    const student = {
      name: 'Александр',
      surname: 'Иванов',
      course: 3,
      age: 21,
      isMarried: false,
      skills: ['HTML', 'CSS', 'JS'],
      address: {
        city: 'Краснодар',
        street: 'Красная',
        house: 22,
        'building/letter': null,
        apartment: 12
      },
    };

    console.log(student.name); //Александр
    console.log(student['name']); //Александр.(name в кавычках, потому что ключ-строковый тип данных)
    console.log(student.address.city); //Краснодар
    console.log(student.address['building/letter']); //null
    console.log(student.skills); //['HTML', 'CSS', 'JS']
    console.log(student.skills[0]); //HTML

    //изменение свойств объекта
    student.isMarried = true;
    console.log(student.isMarried); //true

    //удаление свойств 
    delete student.isMarried;
    console.log(student);

    //добавление свойств
    student.foreignLanguage = 'English';

    //другой пример добавления свойств
    const additionalData = 'foreignLanguage';
    student[additionalData] = 'English';

    //добавление во вложенные свойства
    student.address.country = 'Russia';

    //добавление в массив
    student.skills[3] = 'TypeScript';

    function renderStudentInfo() {
      const studentInfoDiv = document.getElementById('student-info');
      studentInfoDiv.innerHTML = `
        <h2>${student.name} ${student.surname}</h2>
        <p>Курс: ${student.course}</p>
        <p>Возраст: ${student.age}</p>
        <p>Женат: ${student.isMarried ? 'Да' : 'нет'}</p>
        <p>Навыки: ${student.skills.join(", ")}</p>
        <h3>Адрес</h3>
        <p>Город: ${student.address.city}</p>
        <p>Улица: ${student.address.street}</p>
        <p>Дом: ${student.address.house}</p>
        <p>Корпус/Литера: ${
          student.address['building/letter'] === null 
          ? 'Не указано' 
          : student.address['building/letter']
        }</p>
        <p>Квартира: ${student.address.apartment}</p>
      `;
    }

    renderStudentInfo(student);
