Изменение, удаление, добавление элементов

Пример

  
    <!-- HTML -->

    <p>Распределение ролей к утреннику</p>
    <ol id="roles-adults">
      <li class='item'>Дед Мороз - Иван Иванович</li>
      <li class='item'>Снегурочка - Анна Сергеевна</li>
    </ol>
    <p>Снежинки:</p>
    <ul id="roles-children">
      <li class='item'>Маша</li>
      <li class='item'>Света</li>
      <li class='item'>Ира</li>
    </ul>
    <input type="text" id="onemoreChild" placeholder="Введите имя ребёнка" />
    <button id="addChildButton">Добавить снежинку</button>
  

  
    //JS

    //добавление нового имени в список
    const childRolesEl = document.getElementById('roles-children');
    const input = document.getElementById('oneMoreChild');
    const button = document.getElementById('addChildButton');
    //обращаемся к длинне дочерних элементов ненумерованного списка
    let numberChildren = childRolesEl.children.length;

    //добавление в список при нажатии Enter
    input.addEventListener('keypress', (event) => {
      if(event.key === 'Enter') {
        addChild();    
      }
    });

    button.addEventListener('click', () => {
      addChild();
    });

    function addChild() {
      const oneMoreChildInputText = input.value.trim();
      if(oneMoreChildInputText !== '') {
        childRolesEl.innerHTML += `<li class='item'>${oneMoreChildInputText}<button class='delete-button'>Удалить</button></li>`;

        //cоздаём обработчик событий для кнопки удаления
        const deleteButtons = document.querySelectorAll('.delete-button');
        deleteButtons.forEach((btn) => {
          btn.addEventListener('click', () => {
            btn.parentElement.remove(); //parentElement-cвойство для обращения к родительскому элементу
            numberChildren--;  //постфиксный декремент для того чтобы счётчик отнимал значение
            button.disabled = false; //для разблокировки кнопки при значениях меньше 5
          });
        });
      };

      input.value = '';
      numberChildren++;

      if (numberChildren === 5) {
        button.disabled = true;
      }
    }
