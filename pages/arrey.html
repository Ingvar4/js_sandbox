Пример с циклом for

  
    const students = [
      ['Сергей Лобанов', 3, 521301],
      ['Юлия Трофимова', 4, 521401],
      ['Иван Петров', 2, 521201],
      ['Мария Сидорова', 1, 521101],
    ];

    for (const item of students) {
      for (let index = 0; index < item.length; index++) {
        if (index === 0) {
          console.log(`Студент: ${item[index]}`);
        } else if (index === 1) {
          console.log(`\t Курс: ${item[index]}`);
        } else {
          console.log(`\t Группа: ${item[index]}`);
        }
      }
    }

    //Студент: Сергей Лобанов
    //	 Курс: 3
    //	 Группа: 521301
    //Студент: Юлия Трофимова
    //	 Курс: 4
    //	 Группа: 521401
    //Студент: Иван Петров
    //	 Курс: 2
    //	 Группа: 521201
    //Студент: Мария Сидорова
    //	 Курс: 1
    //	 Группа: 521101
