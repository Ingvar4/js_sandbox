Цепочки промисов (Promise Chain)

назад

Пример совокупности асинхронных задач, которые должны выполняться одна за другой

  
    const request = new Promise((resolve, reject) => {
      setTimeout(() => resolve(1), 1000);
    });

    request
      .then((result) => {
        console.log(result);
        return result * 2;
      })
      .then((result) => {
        console.log(result);
        return result * 2;
      })
      .then((result) => {
        console.log(result);
        return result * 2;
      });

    // 1
    // 2
    // 4

В итоге методом then мы передаём данные по цепочке с помощью return и выполняем какую-то работу.

Пример с оценкой производительности скрипта и выполнении работы внутри then

  
    const startAt = Date.now();
    console.log(startAt);

    const request = new Promise((resolve, reject) => {
      setTimeout(() => resolve(1), 0);
    });

    request
      .then((result) => {
        console.log(result);
        return result * 2;
      })
      .then((data) => {
        console.log(data);
        console.log('Время:' + (Date.now() - startAt));
        return data;
      })
      .then((result) => {
        console.log(result);
        return result * 2;
      })
      .then((result) => {
        console.log(result);
        return result * 2;
      })
      .then((data) => {
        console.log(data);
        console.log('Время:' + (Date.now() - startAt));
        return data;
      })
      .then((finalResult) => {
        console.log('Финальный результат:', finalResult);
      });

    // 1774341993342
    // 1
    // 2
    // Время:5
    // 2
    // 4
    // 8
    // Время:5
    // Финальный результат: 8

Время:5 это количество миллисекунд, которые нужны, чтобы скрипт дошел до определённого момента выполнения кода. Задержки в таймауте нет, если добавить секунду то будет Время:1005

Если не делать цепочку, а каждое выражение начинать с request, то никакого умножения на 2 не будет, будут все 1, потому что данные не будут передаваться

Еще пример

  
    const request = new Promise((resolve, reject) => {
      const success = 0.9; // Меняйте значение, например, 0,4, 0.6, 0.9
      if (success > 0.5) {
        resolve(success);
      } else {
        reject("Ошибка на первом этапе");
      }
    });

    request
      .then((result) => {
        console.log("Первый then:", result);
        if (result < 0.7) {
          throw new Error("Ошибка во втором then");
        }
        return result * 2;
      })
      .then((result) => {
        console.log("Второй then:", result);
        return result * 2;
      })
      .catch((error) => {
        console.error("Первый catch:", error);
        return "Значение по умолчанию"; // Возвращаем значение для продолжения
      })
      .then((result) => {
        console.log("Третий then:", result);
        throw new Error("Ошибка в третьем then"); // Пробрасываем новую ошибку
      })
      .catch((error) => {
        console.error("Второй catch:", error.message);
        return "Новое значение по умолчанию"; // Возвращаем значение для продолжения
      })
      .then((result) => {
        console.log("Четвертый then:", result);
      });
