Метод filter()

назад

Пример с сортировкой чётных чисел

  
    const arr = [3, 4, 6, 7, 88, 45, 66];
    const even = arr.filter((el) => el % 2 === 0);
    console.log(even); //[ 4, 6, 88, 66 ]

синтаксис

  
    const newArray = array.filter(callback(element[, index[, array]])[, thisArg]);
  • callback - функция, которая будет вызвана для каждого элемента массива. Она должна возвращать true, чтобы сохранить элемент в новом массиве, или false, чтобы отбросить его.
  • element - текущий обрабатываемый элемент массива.
  • index - (необязательно) индекс текущего обрабатываемого элемента в массиве.
  • array - (необязательно) исходный массив, на котором был вызван метод filter.
  • thisArg (необязательно) значение, используемое в качестве this при выполнении функции callback
  
    const arr = [3, 4, 6, 7, 88, 45, 66];
    const filteredNumbers = arr.filter((num) => num > 10);
    console.log(filteredNumbers); //[ 88, 45, 66 ]
  

  
    const words = ['Рябина', 'Ива', 'Клён', 'Тополь', 'Черёмуха'];
    const longWords = words.filter((word) => word.length > 5);
    console.log(longWords); //[ 'Рябина', 'Тополь', 'Черёмуха' ]
  

  
    const people = [
      {name: 'Alice', age: 25},
      {name: 'Bob', age: 35},
      {name: 'John', age: 12},
    ];

    const adults = people.filter((person) => person.age >= 18);
    console.log(adults); //[ { name: 'Alice', age: 25 }, { name: 'Bob', age: 35 } ]
  

  
    const topics = [
      'JavaScript Basics',
      'Python for Beginners',
      'Advanced JavaScript',
      'PHP for Beginners',
    ];

    const jstopics = topics.filter((topic) => topic.includes('JavaScript'));
    console.log(jstopics); //[ 'JavaScript Basics', 'Advanced JavaScript' ]
