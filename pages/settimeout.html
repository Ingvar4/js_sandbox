Планирование вызова: setTimeout и setInterval

назад

Функция setTimeout позволяет вызывать колбэк один раз через определённый интервал времени

Функция setIntervalпозволяет вызывать колбэк регулярно, повторяя вызов через определённый интервал времени

setTimeout

Синтаксис

  
    setTimeout(function, delay, param1, param2, ...);

    //function: функция, которая будет выполнена после задержки
    //delay: зарежка в миллисекундах (1000мс - 1 сек)
    //param1, param2, ...дополнительные параметры, которые будут переданы в функцию
  

  
    setTimeout(() => {
      console.log('Прошло 5 секунд');
    }, 5000);
  

  
    function logMessage() {
      console.log('Прошло 5 секунд');
    }

    setTimeout(logMessage, 5000);
    console.log('Этот код будет выполнен сразу');

    //Этот код будет выполнен сразу
    //Прошло 5 секунд

Пример с передачей параметров

  
    setTimeout((name, age) => {
      console.log(`Привет, ${name}! Тебе ${age} лет`);
    }, 2000, 'Иван', 30); //Привет, Иван! Тебе 30 лет

Пример с отменой операции через clearTimeout

  
    const timerId = setTimeout(() => {
      console.log('Прошло 5 секунд');
    }, 5000);

    setTimeout(() => {
      clearTimeout(timerId);
      console.log('Таймер отменен');
    }, 3000);

    //Таймер отменен

Пример с отменой таймера через пользовательский ввод

  
    const timerId = setTimeout(() => {
      console.log('Прошло 10 секунд');
    }, 10000);

    function checkForCancel() {
      const userInput = prompt('Введите "отмена" для отмены таймера');

      if (userInput && userInput.toLowerCase() === 'отмена') {
        clearTimeout(timerId);
        console.log('Таймер отменен');
      } else {
        setTimeout(checkForCancel, 1000);
      }
    };

    setTimeout(checkForCancel, 1000);

userInput && userInput.toLowerCase() - такая конструкция, чтобы при нажатии в окне 'Отмена', не вываливалась ошибка, потому что prompt работает с true или false, а у нас userInput приводится к строке, соотв. prompt не работает со строками. Чтобы наше выражение приняло булевые значения мы добавляем userInput &&

setInterval

  
    let timerId = setInterval(() => {
      alert('tick');
    }, 2000);

    setTimeout(() => {
      clearInterval(timerId);
      alert('stop');
    }, 8000);
