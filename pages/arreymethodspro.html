Продвинутый перебор объектов: Object.keys, Object.values, Object. entries

назад

Object.keys()

Используется для получения массива строковых ключей (имен свойств) перечисляемых свойств объекта

  
    const obj = { a: 1, b: 2, c: 3 };
    const keys = Object.keys(obj);
    console.log(keys); //['a', 'b', 'c']
  

  
    const obj = { a: 1, b: 2, c: 3 };
    const keys = Object.keys(obj);
    keys.forEach((key) => {
      console.log(`${key}`);
    });

    //a
    //b
    //c

Так же можно выводить и значения объекта

  
    const obj = { a: 1, b: 2, c: 3 };
    const keys = Object.keys(obj);
    keys.forEach((key) => {
      console.log(`${key}: ${obj[key]}`);
    });

    //a: 1
    //b: 2
    //c: 3

Практический пример

  
    const users = {
      alice: {
        name: 'Alice',
        age: 28,
        email: 'aclice@example.com'
      },
      bob: {
        name: 'Bob',
        age: 34,
        email: 'bob@example.com'
      },
      charlie: {
        name: 'Charlie',
        age: 22,
        email: 'charlie@example.com'
      }
    };

    const keys = Object.keys(users);
    keys.forEach((key) => {
      console.log(`Пользователь: ${users[key].name}`);
      console.log(`Возраст: ${users[key].age}`);
      console.log(`Email: ${users[key].email}`);
      console.log('---');
    });

    // Пользователь: Alice
    // Возраст: 28
    // Email: aclice@example.com
    // ---
    // Пользователь: Bob
    // Возраст: 34
    // Email: bob@example.com
    // ---
    // Пользователь: Charlie
    // Возраст: 22
    // Email: charlie@example.com
    // ---

можно задать переменную user чтобы облегчить чтение кода

  
    const keys = Object.keys(users);
    keys.forEach((key) => {
      const user = users[key];
      console.log(`Пользователь: ${user.name}`);
      console.log(`Возраст: ${user.age}`);
      console.log(`Email: ${user.email}`);
      console.log('---');
    });

Пример если нужно использовать фильтрацию для чего-либо. В данном примере для возраста

  
    const keys = Object.keys(users);
    const filteredKeys = keys.filter((key) => users[key].age > 30);
    filteredKeys.forEach((key) => {
      const user = users[key];
      console.log(`Пользователь: ${user.name}`);
      console.log(`Возраст: ${user.age}`);
      console.log(`Email: ${user.email}`);
      console.log('---');
    });

    // Пользователь: Bob
    // Возраст: 34
    // Email: bob@example.com
    // ---

Object.values()

Используется для получения массива значений перечисляемых свойств объекта

  
    const obj = { a: 1, b: 2, c: 3 };
    const values = Object.values(obj);
    console.log(values); // [1, 2, 3]
  

  
    const products = {
      apple: {
        name: 'яблоки',
        price: 80,
      },
      banana: {
        name: 'бананы',
        price: 100,
      },
      orange: {
        name: 'апельсины',
        price: 150,
      },
    };

    const fruits = Object.values(products);

    fruits.forEach((fruit) => {
      console.log(`Продукт: ${fruit.name}, цена ${fruit.price} руб.`);
    });

    // Продукт: яблоки, цена 80 руб.
    // Продукт: бананы, цена 100 руб.
    // Продукт: апельсины, цена 150 руб.

Object.entries()

Используется для получения массива пар [ключ: значение] перечисляемых свойств объекта

  
    const obj = { a: 1, b: 2, c: 3 };
    const entries = Object.entries(obj);
    console.log(entries); // [['a', 1], ['b', 2], ['c', 3]]
    entries.forEach(([key, value]) => {
      console.log(`${key}: ${value}`);
    });

    // a: 1
    // b: 2
    // c: 3
  

  
    const students = {
      alice: {
        name: 'Alice',
        age: 22,
        grade: 5,
      },
      bob: {
        name: 'Bob',
        age: 24,
        grade: 4,
      },
      charlie: {
        name: 'Charlie',
        age: 21,
        grade: 3,
      }
    };

    const entries = Object.entries(students);
    console.log(entries);
    // Выводит многомерный массив
    // [
    //   [ 'alice', { name: 'Alice', age: 22, grade: 5 } ],
    //   [ 'bob', { name: 'Bob', age: 24, grade: 4 } ],
    //   [ 'charlie', { name: 'Charlie', age: 21, grade: 3 } ]
    // ]

    entries.forEach(([key, student]) => {
      console.log(`ID студента ${key}`);
      console.log(`Имя студента ${student.name}`);
      console.log(`Возраст ${student.age}`);
      console.log(`Оценка ${student.grade}`);
      console.log(`---`);
    })

    // ID студента alice
    // Имя студента Alice
    // Возраст 22
    // Оценка 5
    // ---
    // ID студента bob
    // Имя студента Bob
    // Возраст 24
    // Оценка 4
    // ---
    // ID студента charlie
    // Имя студента Charlie
    // Возраст 21
    // Оценка 3
    // ---

пример с фильтром по оценкам (5)

  
    const students = {
      alice: {
        name: 'Alice',
        age: 22,
        grade: 5,
      },
      bob: {
        name: 'Bob',
        age: 24,
        grade: 4,
      },
      charlie: {
        name: 'Charlie',
        age: 21,
        grade: 3,
      }
    };

    const entries = Object.entries(students);
    const filteredStudents = entries.filter(
      ([, student]) => student.grade === 5
    );

    filteredStudents.forEach(([key, student]) => {
      console.log(`ID студента ${key}`);
      console.log(`Имя студента ${student.name}`);
      console.log(`Возраст ${student.age}`);
      console.log(`Оценка ${student.grade}`);
      console.log(`---`);
    })

    // ID студента alice
    // Имя студента Alice
    // Возраст 22
    // Оценка 5
    // ---
