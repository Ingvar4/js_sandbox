Оператор "..." (spread / rest)

назад

Пример spread

  
    const fruits = ['banana', 'cherry', 'lemon', 'orange', 'pear'];
    const numbers = [3, 4, 12, 6, 8];

    const addedFruits = ['apple', ...fruits, 'plum'];
    const addedNumbers = [5, ...numbers, '7', '88'];

    console.log(addedFruits);
    console.log(addedNumbers);

    // [
    //   'apple',  'banana',
    //   'cherry', 'lemon',
    //   'orange', 'pear',
    //   'plum'
    // ]
    // [
    //   5, 3, 4,   12,
    //   6, 8, '7', '88'
    // ]

Пример с копированием объекта и изменением свойств без изменения исходного объекта

Так происходит, потому что формируется новый объект а не сохранение по ссылке

  
    const person = {
      id: 1,
      name: 'Bob',
      surname: 'Vatson',
      age: 29,
      skills: ['HTML', 'CSS', 'JS', 'PHP'],
    };

    const person2 = { ...person };
    person2.id = 2;
    person2.name = 'Kot';

    console.table(person);
    console.table(person2);

Пример rest

  
    const numbers = [2, 4, 7, 9, 12];

    function print(a, b, ...args) {
      console.log(a, b);
      console.log(args);
    };

    print(...numbers); //2 4
                      //[ 7, 9, 12 ]
  

  
    const numbers = [2, 4, 7, 9, 12];

    function print(a, b, c, ...args) {
      let result = a + b + c;
      console.log(result);
      console.log(args);
    };

    print(...numbers); //13
                      //[ 9, 12 ]
