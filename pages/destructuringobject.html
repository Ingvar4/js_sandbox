Деструктуризация объектов в JavaScript

назад

При деструктуризации объектов уже нельзя давать произвольные названия переменным

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    const { id, isMarried, age, fullName } = person;
    console.log(id, isMarried, age, fullName); //1 true 34 John Jonson

Пример с синтаксисом переименования для тех случаев когда имя нужной переменной для дестр. уже занято

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    let age = 26;

    const { id, isMarried, age: agePerson, fullName } = person;
    console.log(id, isMarried, agePerson, fullName); //1 true 34 John Jonson

Пример если названия переменных переименовывать нельзя. Убираем const и заключаем выражение в круглые скобки

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    let id = 1;
    let age = 26;

    ({ id, isMarried, age, fullName } = person);
    console.log(id, isMarried, age, fullName); //1 true 34 John Jonson

Пример со значениями по умолчанию. Так же можно применить оператор остатка для вычисления остальных свойств

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    const { id = 1, isMarried = false, age = null, ...otherProps } = person;
    console.log(id, isMarried, age, otherProps);

    // 1 true 34 {
    //   fullName: 'John Jonson',
    //   params: { height: 178, weight: 75 },
    //   hobbies: [ 'web', 'jogging', 'cycling' ]
    // }

    //Остаток сам по себе является объектом

Пример с деструктуризацией вложенных объектов

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    const { params: { height, weight } } = person;
    console.log(height, weight); //178 75

Пример с тем, когда нужно извлечь вложенный массив

  
    const person = {
      id: 1,
      fullName: 'John Jonson',
      age: 34,
      isMarried: true,
      params: {
        height: 178,
        weight: 75,
      },
      hobbies: ['web', 'jogging', 'cycling'],
    };

    const { hobbies: [, , lastHobby] } = person;
    console.log(lastHobby); //cycling
